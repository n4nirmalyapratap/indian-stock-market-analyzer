services:
  backend:
    build:
      context: ./artifacts/python-backend
      dockerfile: Dockerfile
    container_name: stock-backend-dev
    restart: unless-stopped
    working_dir: /app
    command: uvicorn main:app --host 0.0.0.0 --port 8090 --reload
    environment:
      PORT: "8090"
      PYTHONPATH: /app
      WATCHFILES_FORCE_POLLING: "true"
      SESSION_SECRET: "${SESSION_SECRET}"
      CLERK_SECRET_KEY: "${CLERK_SECRET_KEY:-}"
      CLERK_PUBLISHABLE_KEY: "${CLERK_PUBLISHABLE_KEY:-}"
      ADMIN_USERNAME: "${ADMIN_USERNAME:-admin}"
      ADMIN_PASSWORD: "${ADMIN_PASSWORD}"
      DATA_DIR: "/app/data"
      TELEGRAM_BOT_TOKEN: "${TELEGRAM_BOT_TOKEN:-}"
      TWILIO_ACCOUNT_SID: "${TWILIO_ACCOUNT_SID:-}"
      TWILIO_AUTH_TOKEN: "${TWILIO_AUTH_TOKEN:-}"
    volumes:
      - ./artifacts/python-backend:/app
      - userdata-dev:/app/data
    ports:
      - "8090:8090"

  user-frontend:
    image: node:24-slim
    container_name: stock-user-frontend-dev
    restart: unless-stopped
    working_dir: /workspace
    command: >
      sh -lc "npm install -g pnpm@10.33.0 &&
      pnpm install --no-frozen-lockfile &&
      PORT=3002 BASE_PATH=/ VITE_API_PROXY_TARGET=http://backend:8090
      pnpm --filter @workspace/stock-market-app run dev"
    environment:
      CHOKIDAR_USEPOLLING: "true"
      CHOKIDAR_INTERVAL: "300"
    volumes:
      - .:/workspace
      - pnpm-store:/root/.local/share/pnpm/store
      - user-workspace-node-modules:/workspace/node_modules
    depends_on:
      - backend
    ports:
      - "3002:3002"

  admin-frontend:
    image: node:24-slim
    container_name: stock-admin-frontend-dev
    restart: unless-stopped
    working_dir: /workspace
    command: >
      sh -lc "npm install -g pnpm@10.33.0 &&
      pnpm install --no-frozen-lockfile &&
      PORT=22133 BASE_PATH=/admin/ VITE_API_PROXY_TARGET=http://backend:8090
      pnpm --filter @workspace/admin-dashboard run dev"
    environment:
      CHOKIDAR_USEPOLLING: "true"
      CHOKIDAR_INTERVAL: "300"
    volumes:
      - .:/workspace
      - pnpm-store:/root/.local/share/pnpm/store
      - admin-workspace-node-modules:/workspace/node_modules
    depends_on:
      - backend
    ports:
      - "22133:22133"

  proxy:
    image: nginx:alpine
    container_name: stock-dev-proxy
    restart: unless-stopped
    depends_on:
      - backend
      - user-frontend
      - admin-frontend
    volumes:
      - ./artifacts/stock-market-app/nginx.dev.conf:/etc/nginx/conf.d/default.conf:ro
    ports:
      - "80:80"

volumes:
  userdata-dev:
  pnpm-store:
  user-workspace-node-modules:
  admin-workspace-node-modules:
