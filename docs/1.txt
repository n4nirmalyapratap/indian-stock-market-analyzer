pythonimport requests
import pandas as pd
import time
from datetime import datetime, timedelta

HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                  "AppleWebKit/537.36 (KHTML, like Gecko) "
                  "Chrome/120.0.0.0 Safari/537.36",
    "Accept-Language": "en-US,en;q=0.9",
    "Referer": "https://www.nseindia.com/",
}

# ─────────────────────────────────────────────
# STEP 1: Create session (mandatory for NSE)
# ─────────────────────────────────────────────
def get_nse_session():
    session = requests.Session()
    session.headers.update(HEADERS)
    session.get("https://www.nseindia.com", timeout=10)
    time.sleep(1)  # Let cookies settle
    return session


# ─────────────────────────────────────────────
# TAB 1: EQUITY  
# ─────────────────────────────────────────────
def get_equity_fii_dii(session, start_date: str, end_date: str) -> pd.DataFrame:
    """
    Fetch Equity FII/DII data.
    Dates format: 'DD-MM-YYYY'  e.g. '01-04-2026'
    Returns DataFrame with: date, fii_buy, fii_sell, fii_net, dii_buy, dii_sell, dii_net
    """
    url = (
        f"https://www.nseindia.com/api/historical/fiidii"
        f"?startDate={start_date}&endDate={end_date}"
    )
    resp = session.get(url, timeout=10)
    raw = resp.json()

    # NSE wraps in {"data": [...]} or returns list directly
    rows = raw.get("data", raw) if isinstance(raw, dict) else raw

    records = []
    for r in rows:
        records.append({
            "date":      r.get("date"),
            "fii_buy":   float(r.get("buyValue",  r.get("fiiBuyVal",  0)) or 0),
            "fii_sell":  float(r.get("sellValue", r.get("fiiSellVal", 0)) or 0),
            "fii_net":   float(r.get("netValue",  r.get("fiiNetVal",  0)) or 0),
            "dii_buy":   float(r.get("diiBuyVal",  0) or 0),
            "dii_sell":  float(r.get("diiSellVal", 0) or 0),
            "dii_net":   float(r.get("diiNetVal",  0) or 0),
        })
    return pd.DataFrame(records)


# ─────────────────────────────────────────────
# TABS 2-5: INDEX FUTURE / INDEX OPTION /
#           STOCK FUTURE / STOCK OPTIONS
# ─────────────────────────────────────────────

# Mapping: tab name → NSE internal segment key
SEGMENT_MAP = {
    "index_future":  "FUTIDX",   # Index Futures
    "index_option":  "OPTIDX",   # Index Options
    "stock_future":  "FUTSTK",   # Stock Futures
    "stock_options": "OPTSTK",   # Stock Options
}

def get_fno_participant_data(
    session,
    segment: str,          # one of SEGMENT_MAP keys
    start_date: str,
    end_date: str
) -> pd.DataFrame:
    """
    Fetch F&O participant-wise data for any segment.
    segment: 'index_future' | 'index_option' | 'stock_future' | 'stock_options'
    Dates: 'DD-MM-YYYY'
    Returns DataFrame with FII columns (buy_contracts, sell_contracts, net_contracts,
    buy_value, sell_value, net_value) per date.
    """
    seg_code = SEGMENT_MAP[segment]
    url = (
        f"https://www.nseindia.com/api/historical/fnoparticipants"
        f"?startDate={start_date}&endDate={end_date}"
    )
    resp = session.get(url, timeout=10)
    raw = resp.json()
    rows = raw.get("data", raw) if isinstance(raw, dict) else raw

    records = []
    for r in rows:
        # NSE returns rows for ALL segments in one call.
        # Each row has a key like 'futIdxFiiLong', 'optIdxFiiLong', etc.
        # We filter by segment prefix:
        prefix_map = {
            "FUTIDX": "futIdx",
            "OPTIDX": "optIdx",
            "FUTSTK": "futStk",
            "OPTSTK": "optStk",
        }
        p = prefix_map[seg_code]

        records.append({
            "date":              r.get("date"),
            # Contracts (number of contracts bought/sold)
            "fii_long":          _f(r, f"{p}FiiLong"),
            "fii_short":         _f(r, f"{p}FiiShort"),
            "fii_net_contracts": _f(r, f"{p}FiiLong") - _f(r, f"{p}FiiShort"),
            "dii_long":          _f(r, f"{p}DiiLong"),
            "dii_short":         _f(r, f"{p}DiiShort"),
            "dii_net_contracts": _f(r, f"{p}DiiLong") - _f(r, f"{p}DiiShort"),
            "client_long":       _f(r, f"{p}ClientLong"),
            "client_short":      _f(r, f"{p}ClientShort"),
            "pro_long":          _f(r, f"{p}ProLong"),
            "pro_short":         _f(r, f"{p}ProShort"),
        })
    return pd.DataFrame(records)

def _f(row, key):
    return float(row.get(key, 0) or 0)


# ─────────────────────────────────────────────
# MASTER FUNCTION: fetch all 5 tabs at once
# ─────────────────────────────────────────────
def get_all_tabs(days: int = 30) -> dict:
    """
    Returns a dict with DataFrames for all 5 tabs.
    {
      "equity":        DataFrame,
      "index_future":  DataFrame,
      "index_option":  DataFrame,
      "stock_future":  DataFrame,
      "stock_options": DataFrame,
    }
    """
    end   = datetime.today()
    start = end - timedelta(days=days)
    start_str = start.strftime("%d-%m-%Y")
    end_str   = end.strftime("%d-%m-%Y")

    session = get_nse_session()

    result = {}

    # Tab 1 — Equity
    result["equity"] = get_equity_fii_dii(session, start_str, end_str)
    time.sleep(1)

    # Tabs 2–5 — All F&O segments
    for segment in SEGMENT_MAP:
        result[segment] = get_fno_participant_data(session, segment, start_str, end_str)
        time.sleep(1)   # Be polite to NSE servers

    return result

pythonimport requests
import pandas as pd
import time
import json
import os
import sqlite3
from datetime import datetime, timedelta
from typing import Literal

# ─────────────────────────────────────────────
# CONFIG
# ─────────────────────────────────────────────
HEADERS = {
    "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                  "AppleWebKit/537.36 (KHTML, like Gecko) "
                  "Chrome/120.0.0.0 Safari/537.36",
    "Accept-Language": "en-US,en;q=0.9",
    "Referer": "https://www.nseindia.com/",
}

NSE_CHUNK_DAYS = 85          # NSE breaks if you go beyond ~90 days per call
DB_FILE = "fii_dii_cache.db" # Local SQLite cache

SEGMENT_MAP = {
    "equity":        None,       # Uses different endpoint
    "index_future":  "futIdx",
    "index_option":  "optIdx",
    "stock_future":  "futStk",
    "stock_options": "optStk",
}

SegmentType = Literal["equity", "index_future", "index_option", "stock_future", "stock_options"]


# ─────────────────────────────────────────────
# SESSION
# ─────────────────────────────────────────────
def get_nse_session() -> requests.Session:
    session = requests.Session()
    session.headers.update(HEADERS)
    session.get("https://www.nseindia.com", timeout=10)
    time.sleep(1)
    return session


# ─────────────────────────────────────────────
# DATE CHUNKER — splits long ranges into 85-day pieces
# ─────────────────────────────────────────────
def date_chunks(start: datetime, end: datetime, chunk_days: int = NSE_CHUNK_DAYS):
    """
    Yields (chunk_start, chunk_end) pairs of max chunk_days each.
    Example: 1 year range → 5 chunks of ~85 days each.
    """
    current = start
    while current < end:
        chunk_end = min(current + timedelta(days=chunk_days), end)
        yield current, chunk_end
        current = chunk_end + timedelta(days=1)
        time.sleep(1.2)   # polite delay between NSE calls


# ─────────────────────────────────────────────
# RETRY WRAPPER
# ─────────────────────────────────────────────
def safe_get(session: requests.Session, url: str, retries: int = 3) -> dict | list | None:
    for attempt in range(retries):
        try:
            resp = session.get(url, timeout=12)
            if resp.status_code == 403:
                print(f"  [403] Refreshing session... (attempt {attempt+1})")
                session = get_nse_session()   # Refresh cookies
                continue
            if resp.status_code == 429:
                wait = 2 ** attempt
                print(f"  [429] Rate limited. Waiting {wait}s...")
                time.sleep(wait)
                continue
            resp.raise_for_status()
            return resp.json()
        except Exception as e:
            print(f"  Error: {e} (attempt {attempt+1}/{retries})")
            time.sleep(2)
    return None


# ─────────────────────────────────────────────
# PARSE RAW NSE RESPONSE → clean dict
# ─────────────────────────────────────────────
def parse_equity_row(r: dict) -> dict:
    return {
        "date":      r.get("date"),
        "fii_buy":   _f(r, "buyValue")   or _f(r, "fiiBuyVal"),
        "fii_sell":  _f(r, "sellValue")  or _f(r, "fiiSellVal"),
        "fii_net":   _f(r, "netValue")   or _f(r, "fiiNetVal"),
        "dii_buy":   _f(r, "diiBuyVal"),
        "dii_sell":  _f(r, "diiSellVal"),
        "dii_net":   _f(r, "diiNetVal"),
    }

def parse_fno_row(r: dict, prefix: str) -> dict:
    long_  = _f(r, f"{prefix}FiiLong")
    short_ = _f(r, f"{prefix}FiiShort")
    d_long  = _f(r, f"{prefix}DiiLong")
    d_short = _f(r, f"{prefix}DiiShort")
    return {
        "date":              r.get("date"),
        "fii_long":          long_,
        "fii_short":         short_,
        "fii_net":           long_ - short_,
        "dii_long":          d_long,
        "dii_short":         d_short,
        "dii_net":           d_long - d_short,
        "client_long":       _f(r, f"{prefix}ClientLong"),
        "client_short":      _f(r, f"{prefix}ClientShort"),
        "pro_long":          _f(r, f"{prefix}ProLong"),
        "pro_short":         _f(r, f"{prefix}ProShort"),
    }

def _f(row: dict, key: str) -> float:
    return float(row.get(key, 0) or 0)


# ─────────────────────────────────────────────
# FETCH WITH CHUNKING — works for any date range
# ─────────────────────────────────────────────
def fetch_equity_historical(
    session: requests.Session,
    start: datetime,
    end: datetime
) -> pd.DataFrame:
    all_rows = []
    for cs, ce in date_chunks(start, end):
        s = cs.strftime("%d-%m-%Y")
        e = ce.strftime("%d-%m-%Y")
        print(f"  Fetching Equity: {s} → {e}")
        url  = f"https://www.nseindia.com/api/historical/fiidii?startDate={s}&endDate={e}"
        data = safe_get(session, url)
        if data is None:
            continue
        rows = data.get("data", data) if isinstance(data, dict) else data
        all_rows.extend([parse_equity_row(r) for r in rows])

    df = pd.DataFrame(all_rows)
    if not df.empty:
        df["date"] = pd.to_datetime(df["date"], dayfirst=True)
        df = df.sort_values("date").drop_duplicates("date").reset_index(drop=True)
    return df


def fetch_fno_historical(
    session: requests.Session,
    segment: SegmentType,
    start: datetime,
    end: datetime
) -> pd.DataFrame:
    prefix = SEGMENT_MAP[segment]
    all_rows = []
    for cs, ce in date_chunks(start, end):
        s = cs.strftime("%d-%m-%Y")
        e = ce.strftime("%d-%m-%Y")
        print(f"  Fetching {segment}: {s} → {e}")
        url  = f"https://www.nseindia.com/api/historical/fnoparticipants?startDate={s}&endDate={e}"

# ─────────────────────────────────────────────
# SQLITE CACHE — avoid re-fetching same dates
# ─────────────────────────────────────────────
def save_to_db(df: pd.DataFrame, table: str):
    if df.empty:
        return
    conn = sqlite3.connect(DB_FILE)
    df.to_sql(table, conn, if_exists="replace", index=False)
    conn.close()
    print(f"  ✅ Saved {len(df)} rows to table '{table}'")

def load_from_db(table: str) -> pd.DataFrame | None:
    if not os.path.exists(DB_FILE):
        return None
    try:
        conn = sqlite3.connect(DB_FILE)
        df = pd.read_sql(f"SELECT * FROM {table}", conn)
        conn.close()
        df["date"] = pd.to_datetime(df["date"])
        return df
    except Exception:
        return None

def get_cached_date_range(table: str):
    """Returns (min_date, max_date) already cached, or (None, None)."""
    df = load_from_db(table)
    if df is None or df.empty:
        return None, None
    return df["date"].min(), df["date"].max()


# ─────────────────────────────────────────────
# SMART FETCH — only downloads missing date ranges
# ─────────────────────────────────────────────
def get_historical(
    segment: SegmentType = "equity",
    start_date: str = None,   # "YYYY-MM-DD" or None for 1 year back
    end_date:   str = None,   # "YYYY-MM-DD" or None for today
    force_refresh: bool = False
) -> pd.DataFrame:
    """
    Main function. Fetches FII/DII historical data for any segment.
    
    segment:      'equity' | 'index_future' | 'index_option' | 
                  'stock_future' | 'stock_options'
    start_date:   "2023-01-01"  (default: 1 year ago)
    end_date:     "2024-01-01"  (default: today)
    force_refresh: ignore cache and re-fetch everything

    Returns a clean pandas DataFrame sorted by date.
    """
    end   = datetime.strptime(end_date,   "%Y-%m-%d") if end_date   else datetime.today()
    start = datetime.strptime(start_date, "%Y-%m-%d") if start_date else end - timedelta(days=365)

    table = f"fii_dii_{segment}"

    # Check cache
    if not force_refresh:
        cached_min, cached_max = get_cached_date_range(table)
        if cached_min and cached_max:
            # Only fetch what's missing
            fetch_ranges = []
            if start < cached_min:
                fetch_ranges.append((start, cached_min - timedelta(days=1)))
            if end > cached_max:
                fetch_ranges.append((cached_max + timedelta(days=1), end))

            if not fetch_ranges:
                print(f"  📦 Serving '{segment}' from cache ({cached_min.date()} → {cached_max.date()})")
                df = load_from_db(table)
                return df[(df["date"] >= pd.Timestamp(start)) & (df["date"] <= pd.Timestamp(end))]

            # Fetch only the missing chunks
            session = get_nse_session()
            new_dfs = []
            for rs, re in fetch_ranges:
                if segment == "equity":
                    new_dfs.append(fetch_equity_historical(session, rs, re))
                else:
                    new_dfs.append(fetch_fno_historical(session, segment, rs, re))

            # Merge with cache
            cached_df = load_from_db(table)
            combined = pd.concat([cached_df] + new_dfs, ignore_index=True)
            combined = combined.sort_values("date").drop_duplicates("date").reset_index(drop=True)
            save_to_db(combined, table)
            return combined[(combined["date"] >= pd.Timestamp(start)) & (combined["date"] <= pd.Timestamp(end))]

    # Fresh fetch (no cache or force_refresh)
    session = get_nse_session()
    print(f"\n🔄 Fetching {segment} from {start.date()} to {end.date()}...")
    if segment == "equity":
        df = fetch_equity_historical(session, start, end)
    else:
        df = fetch_fno_historical(session, segment, start, end)

    save_to_db(df, table)
    return df

# CONVENIENCE HELPERS
# ─────────────────────────────────────────────
def get_all_segments_historical(

    for seg, df in data.items():
